# Merchant Risk Sentinel — Phase 1 dependency constraints.

# Exact resolved versions are frozen in requirements.lock.txt (Dev Plan §33.4).
#
# pandas is pinned below 3.0 deliberately: the Fraud Detection Handbook raw .pkl files
# were written by pandas 1.x and reference pandas.core.indexes.numeric.Int64Index, which
# was removed in pandas 2.0. pandas 2.x still ships pandas.compat.pickle_compat, which
# maps that legacy class back to pandas.Index when reading. See
# src/mrs/data/legacy_pickle.py for the read path and its fallback.

pandas>=2.2,<3
numpy>=1.26,<3
pyarrow>=15
requests>=2.31
pytest>=8.0

# Phase 4 (ML baseline): scikit-learn for the Logistic Regression baseline pipeline
# (imputation, scaling, model). scipy/joblib are scikit-learn's own runtime deps, pinned
# explicitly so lockfile regeneration is deterministic.
scikit-learn>=1.5,<2
scipy>=1.13,<2
joblib>=1.4,<2

# Phase 5 (main model): XGBoost. On macOS, the xgboost wheel's native library requires
# the OpenMP runtime (Homebrew `libomp`) to be present on the host machine -- a system
# package, not a Python dependency, so it is documented here rather than pinned in this
# file. See docs/MODEL_REPORT.md.
xgboost>=2.0,<3
